# ═══ EC2: Servidores virtuales ═══

# Crear key pair para SSH
aws ec2 create-key-pair \
  --key-name workshop-key \
  --query 'KeyMaterial' \
  --output text > workshop-key.pem

cat workshop-key.pem | head -5

# Crear security group (firewall) y capturar el ID
SG_ID=$(aws ec2 create-security-group \
  --group-name workshop-sg \
  --description "Security group del workshop" \
  --query 'GroupId' --output text)
echo "Security Group ID: $SG_ID"

# Permitir SSH (puerto 22)
aws ec2 authorize-security-group-ingress \
  --group-id $SG_ID\
  --protocol tcp \
  --port 22 \
  --cidr 0.0.0.0/0

# Levantar instancia
aws ec2 run-instances \
  --image-id ami-12345678 \
  --instance-type t2.micro \
  --key-name workshop-key \
  --security-groups workshop-sg \
  --user-data '#!/bin/bash
echo "Hola desde EC2 en Floci!" > /tmp/bienvenida.txt
date >> /tmp/bienvenida.txt'

# Ver instancias
aws ec2 describe-instances \
  --query 'Reservations[].Instances[].{ID:InstanceId,State:State.Name,Type:InstanceType}'

# Guardar ID
INSTANCE_ID=$(aws ec2 describe-instances \
  --query 'Reservations[0].Instances[0].InstanceId' --output text)
echo "Instance ID: $INSTANCE_ID"

# Ver estado
aws ec2 describe-instance-status --instance-ids $INSTANCE_ID

# Detener
aws ec2 stop-instances --instance-ids $INSTANCE_ID

# Verificar estado
aws ec2 describe-instances \
  --instance-ids $INSTANCE_ID \
  --query 'Reservations[].Instances[].State.Name'

# Reiniciar
aws ec2 start-instances --instance-ids $INSTANCE_ID
