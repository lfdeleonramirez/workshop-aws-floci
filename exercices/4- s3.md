# ═══ S3: Almacenamiento ═══

# Crear bucket
aws s3 mb s3://empresa-workshop

# Verificar
aws s3 ls

# Crear archivos de ejemplo
echo "Reporte de ventas Q1 2026" > reporte-q1.txt
echo "Configuración del sistema" > config.json
echo "Notas de la reunión" > notas.txt

# Subir organizados por prefijo
aws s3 cp reporte-q1.txt s3://empresa-workshop/reportes/q1.txt
aws s3 cp config.json s3://empresa-workshop/config.json
aws s3 cp notas.txt s3://empresa-workshop/docs/notas.txt

# Listar todo
aws s3 ls s3://empresa-workshop/ --recursive

# Descargar
aws s3 cp s3://empresa-workshop/reportes/q1.txt descarga.txt
cat descarga.txt

# Sync: sincronizar directorio completo
mkdir proyecto
echo "<h1>Hola</h1>" > proyecto/index.html
echo "body{}" > proyecto/styles.css
echo "console.log('ok')" > proyecto/app.js

aws s3 sync proyecto/ s3://empresa-workshop/website/

# Verificar
aws s3 ls s3://empresa-workshop/website/

# Eliminar un objeto
aws s3 rm s3://empresa-workshop/docs/notas.txt

# Eliminar un prefijo completo
aws s3 rm s3://empresa-workshop/website/ --recursive

# Ver qué quedó
aws s3 ls s3://empresa-workshop/ --recursive
